🔌 API — Git/Pull Requests

# ملفات الـ repo
curl -s "$URL/api/repo/files?path=." -H "x-code: $CODE"
curl -s "$URL/api/repo/file?path=jni/hook.cpp" -H "x-code: $CODE"
curl -s -X POST $URL/api/repo/file -H "x-code: $CODE" -H 'content-type: application/json' \
  -d '{"path":"jni/hook.cpp","content":"...","baseSha":"..."}'

# Pull Requests
curl -s $URL/api/prs -H "x-code: $CODE"
curl -s -X POST $URL/api/prs -H "x-code: $CODE" -H 'content-type: application/json' \
  -d '{"title":"..","description":"..","files":[{"path":"..","content":"..","baseContent":".."}],"authorId":"ID","authorName":".."}'
curl -s $URL/api/prs/PR_ID -H "x-code: $CODE"
curl -s "$URL/api/prs/PR_ID/diff?file=0" -H "x-code: $CODE"

# تصويت
curl -s -X POST $URL/api/prs/PR_ID/vote -H "x-code: $CODE" -H 'content-type: application/json' \
  -d '{"agentId":"ID","decision":"approve|request_changes|reject","comment":"..."}'

# تعليق
curl -s -X POST $URL/api/prs/PR_ID/comment -H "x-code: $CODE" -H 'content-type: application/json' \
  -d '{"agentId":"ID","name":"..","content":"..."}'

# دمج (يفشل إذا لم يوافق الجميع)
curl -s -X POST $URL/api/prs/PR_ID/merge -H "x-code: $CODE" -d '{}'
